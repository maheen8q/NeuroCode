cmake_minimum_required(VERSION 3.18)

project(nc_train CXX)
enable_testing()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

# Find required packages
find_package(Torch REQUIRED)
find_package(nlohmann_json 3 REQUIRED)

set(SRC ${CMAKE_CURRENT_SOURCE_DIR}/../../src)

# ML module
add_library(nc_ml STATIC
    ${SRC}/ml/Dataset.cpp
    ${SRC}/ml/Model.cpp
    ${SRC}/ml/Metrics.cpp
    ${SRC}/ml/Trainer.cpp
    ${SRC}/ml/StubPredictor.cpp
    ${SRC}/ml/Predictor.cpp
)

target_include_directories(nc_ml PUBLIC ${SRC})

target_link_libraries(nc_ml
    PUBLIC
    "${TORCH_LIBRARIES}"
    nlohmann_json::nlohmann_json
)

# Test executables
foreach(t
    test_dataset
    test_standardize
    test_tensors
    test_model
    test_metrics
    test_train
    test_save
    test_predictor
    test_cv
)
    add_executable(${t} ${t}.cpp)
    target_link_libraries(${t} PRIVATE nc_ml)

    add_test(NAME ${t} COMMAND ${t})

    set_tests_properties(${t} PROPERTIES
        WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../.."
    )
endforeach()

# Training executable
add_executable(nc_train train_main.cpp)

target_link_libraries(nc_train PRIVATE nc_ml)
